## ------------------------------------------------------------
## 1. Define libraries
## ------------------------------------------------------------

local_lib <- file.path(Sys.getenv("LOCALAPPDATA"), "R", 
                       paste0("win-library/", getRversion()[1,1:2]))
external_lib <- "D:/OneDrive - University of Leeds/MyWorkArea/R"

## Ensure both exist
dir.create(local_lib, recursive = TRUE, showWarnings = FALSE)
dir.create(external_lib, recursive = TRUE, showWarnings = FALSE)

## ------------------------------------------------------------
## 2. Set library paths (local first!)
## ------------------------------------------------------------

.libPaths(c(local_lib, external_lib))

## ------------------------------------------------------------
## 3. Install CRAN packages into external library
## ------------------------------------------------------------

install.packages(
  c("tidyverse", "data.table", "ggplot2"),
  lib = external_lib
)

## ------------------------------------------------------------
## 4. Install Bioconductor core packages into local library
## ------------------------------------------------------------

BiocManager::install(
  c("S4Vectors", "IRanges", "GenomicRanges",
    "SummarizedExperiment", "DelayedArray", "SparseArray"),
  lib = local_lib,
  update = TRUE
)

## ------------------------------------------------------------
## 5. Install DESeq2 (also into local library)
## ------------------------------------------------------------

BiocManager::install("DESeq2", lib = local_lib)

## ------------------------------------------------------------
## 6. Load DESeq2 normally
## ------------------------------------------------------------

library(DESeq2)
