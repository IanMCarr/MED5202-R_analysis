# Installing Bioconductor's BiocManager and DESeq2

if (!requireNamespace("BiocManager", quietly = TRUE)) {

library(BiocManager)

if (!requireNamespace("DESeq2", quietly = TRUE)) {
BiocManager::install(c("DESeq2", "GenomicRanges"), lib = library)
}
library(DESeq2)

#Setting R's work folder
dir.create(workFolder, recursive = TRUE, showWarnings = FALSE)
setwd(workFolder)

# making a variable that points to the read count file

rSubReadFile <- file.path(workFolder, "gene_counts.txt")

# Reading the read count file and prpepaing its data for use

counts <- read.delim(rSubReadFile)
columnsFromDataFile <- c(16, 17, 18, 19, 20, 21)
counts <- as.matrix(counts[, columnsFromDataFile])
storage.mode(counts) <- "integer"
counts <- counts[rowSums(counts > 3) > 2, ]

#Describing the samples and analysis

groupsForEachSample <- c("siCK", "siCK", "siCK", "Control", "Control", "Control")
sampleNames <- c("siCK4_1", "siCK4_2", "siCK4_3", "siNT3_1", "siNT3_2", "siNT3_3")
conditionLevels <- c("Control", "siCK")
sampleTable <- data.frame(sample = sampleNames, condition = groupsForEachSample)
colnames(counts) <- sampleTable$sample
rownames(sampleTable) <- sampleTable$sample

#Setting up and prefoming the DESeq2 analysis

dds <- DESeqDataSetFromMatrix(countData = counts, colData = sampleTable, design = ~ condition)
dds$condition <- factor(dds$condition, levels = conditionLevels)
mcols(dds)$gene <- rownames(counts)
dds <- DESeq(dds)
res <- results(dds)
summary(res)

#Filtering and saving data to file

write.csv(as.data.frame(res), file = "standard_analysis_Deseq2.csv")
resSig <- subset(res, padj < 0.05)
write.csv(as.data.frame(resSig), file = "standard_analysis_Deseq2_significant_05.csv")
